# Función para transformar de vuelta
xform <- function(in.vec, out.vec, val){
  i <- 1
  while (i < length(in.vec) & val >= in.vec[i + 1]) i <- i + 1
  if (i == length(in.vec)) return(out.vec[i])
  return(out.vec[i] + ((val - in.vec[i]) / (in.vec[i + 1] - in.vec[i])) * (out.vec[i + 1] - out.vec[i]))
}

for (j in grep("sim", colnames(Z.sim_data))) {
  Z.sim_data[[j]] <- sapply(Z.sim_data[[j]], function(z_val) {
    xform(data_zinc_ord_sf$Z, data_zinc_ord_sf$zinc, z_val)
  })
}

# Crear un vector para almacenar el conteo de cada simulación

  # Extraer los valores de píxeles para la i-ésima simulación
  sim_values <- Z.sim_data[[paste0("sim", i)]]
  
  # Contar cuántos valores están por encima de 1000
  count <- sum(sim_values > 1000, na.rm = TRUE)
  
  # Almacenar el conteo en el vector
  muy_alto_nivel_counts[i] <- count
}
